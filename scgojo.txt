local function setStatus(text, color)
	status.Text = "●  " .. text
	status.TextColor3 = color or COLORS.Muted
end

local function setProgress(current, target)
	if target <= 0 then
		progress.Size = UDim2.new(0, 0, 1, 0)
		return
	end

	local ratio = math.clamp(current / target, 0, 1)

	TweenService:Create(
		progress,
		TweenInfo.new(0.1, Enum.EasingStyle.Quad),
		{Size = UDim2.new(ratio, 0, 1, 0)}
	):Play()
end

--========================================================--
-- FIND PLAYER
--========================================================--

local function getTarget()
	local text = targetBox.Text
	text = text:gsub("^%s+", ""):gsub("%s+$", ""):lower()

	if text == "" then
		return nil
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if player.Name:lower() == text then
			return player
		end
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if player.DisplayName:lower() == text then
			return player
		end
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if player.Name:lower():sub(1, #text) == text then
			return player
		end
	end

	return nil
end

--========================================================--
-- BULK ADD
--========================================================--

local function startAutoAdd()
	if running then
		return
	end

	local itemKey = itemBox.Text
	itemKey = itemKey:gsub("^%s+", ""):gsub("%s+$", "")

	local targetAmount = tonumber(amountBox.Text)

	if itemKey == "" then
		setStatus("ITEM KEY IS EMPTY", COLORS.Red)
		return
	end

	if not targetAmount then
		setStatus("AMOUNT MUST BE A NUMBER", COLORS.Red)
		return
	end

	targetAmount = math.floor(targetAmount)

	if targetAmount <= 0 then
		setStatus("AMOUNT MUST BE ABOVE 0", COLORS.Red)
		return
	end

	running = true

	setProgress(0, targetAmount)

	setStatus(
		"ADDING " .. itemKey .. " • " .. targetAmount,
		COLORS.Cyan
	)

	task.spawn(function()

		--================================================--
		-- BULK QUANTITY
		--================================================--
		-- Server sekarang menerima quantity langsung.
		-- Jadi 300 item = 1 FireServer, bukan 300 request.

		ChangeOffer:FireServer(itemKey, targetAmount)

		if not running then
			return
		end

		setProgress(targetAmount, targetAmount)

		setStatus(
			"COMPLETED • " ..
			itemKey ..
			" x" ..
			targetAmount,
			COLORS.Green
		)

		running = false
	end)
end

autoButton.MouseButton1Click:Connect(startAutoAdd)

-- TRADE EVENTS
--========================================================--

TradeEvent.OnClientEvent:Connect(function(eventName, data)

	if eventName == "RequestReceived" then

		if data and data.player then
			setStatus(
				"INCOMING FROM " .. data.player.DisplayName,
				COLORS.Yellow
			)
		else
			setStatus(
				"INCOMING TRADE REQUEST",
				COLORS.Yellow
			)
		end

	elseif eventName == "RequestSent" then

		setStatus(
			"REQUEST SENT",
			COLORS.Cyan
		)

	elseif eventName == "RequestClosed" then

		setStatus(
			"REQUEST CLOSED",
			COLORS.Muted
		)

	elseif eventName == "RequestExpired" then

		setStatus(
			"REQUEST EXPIRED",
			COLORS.Red
		)

	elseif eventName == "Started" then

		running = false

		if data and data.partner then
			currentTarget = data.partner

			setStatus(
				"TRADE STARTED • " ..
				data.partner.DisplayName,
				COLORS.Green
			)
		else
			setStatus(
				"TRADE STARTED",
				COLORS.Green
			)
		end

	elseif eventName == "Updated" then

		if data and data.phase then

			local phase = tostring(data.phase)

			if phase == "Countdown" then

				setStatus(
					"COUNTDOWN...",
					COLORS.Yellow
				)

			elseif phase == "Offer" then

				setStatus(
					"OFFER PHASE",
					COLORS.Cyan
				)

			elseif phase == "Confirm" then

				setStatus(
					"CONFIRM PHASE",
					COLORS.Yellow
				)

			else

				setStatus(
					phase:upper(),
					COLORS.Muted
				)

			end
		end

	elseif eventName == "Ended" then

		running = false

		if data and data.reason then
			setStatus(
				"TRADE ENDED • " ..
				tostring(data.reason),
				COLORS.Red
			)
		else
			setStatus(
				"TRADE ENDED",
				COLORS.Red
			)
		end
	end
end)

--========================================================--
-- DRAG
--========================================================--

local dragging = false
local dragStart
local startPosition

header.InputBegan:Connect(function(input)

	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPosition = main.Position

		input.Changed:Connect(function()

			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end

		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)

	if not dragging then
		return
	end

	if input.UserInputType ~= Enum.UserInputType.MouseMovement
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	local delta = input.Position - dragStart

	main.Position = UDim2.new(
		startPosition.X.Scale,
		startPosition.X.Offset + delta.X,
		startPosition.Y.Scale,
		startPosition.Y.Offset + delta.Y
	)
end)
